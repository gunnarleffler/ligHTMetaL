  return greeting + name; // Lexical highlighting
}

TypeScript

light

interface Person {
  readonly name: string;
  age?: number;
}

type Maybe<T> = T | null;
function identity<T>(value: T): T {
  return value;
}
const person: Maybe<Person> = identity<Person>({ name: "Ada" });
